"use strict"; // ============================================ // EL JARDÍN DE SNOOPY // Carrito y pedidos de demostración // ============================================ const carrito = []; const PRECIOS_TAMANO = { "Pequeño": 250, "Mediano": 400, "Grande": 600 }; const PRECIOS_ENVOLTURA = { "Papel kraft": 0, "Papel rosa": 30, "Papel elegante": 50, "Envoltura premium": 80 }; const formatoDinero = cantidad => Number(cantidad).toLocaleString("es-MX", { minimumFractionDigits: 0, maximumFractionDigits: 2 }); const obtenerElemento = id => document.getElementById(id); function establecerTexto(id, texto) { const elemento = obtenerElemento(id); if (elemento) elemento.textContent = texto; } function escaparHTML(texto) { return String(texto ?? "").replace(/[&<>"']/g, caracter => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[caracter]); } function obtenerTotal() { return carrito.reduce( (suma, producto) => suma + producto.precio * producto.cantidad, 0 ); } // ============================================ // AGREGAR PRODUCTOS AL CARRITO // ============================================ function agregar(nombre, precio) { precio = Number(precio); if (!nombre || !Number.isFinite(precio) || precio < 0) { return; } const existente = carrito.find(producto => producto.nombre === nombre && !producto.personalizado ); if (existente) { existente.cantidad++; } else { carrito.push({ nombre, precio, cantidad: 1, personalizado: false }); } actualizarCarrito(); mostrarMensaje(`${nombre} se agregó al carrito.`); } function cambiarCantidad(indice, cambio) { const producto = carrito[indice]; if (!producto) return; producto.cantidad += cambio; if (producto.cantidad <= 0) { carrito.splice(indice, 1); } actualizarCarrito(); } function eliminarProducto(indice) { if (indice < 0 || indice >= carrito.length) return; carrito.splice(indice, 1); actualizarCarrito(); } function actualizarCarrito() { const lista = obtenerElemento("lista-carrito"); if (!lista) return; if (carrito.length === 0) { lista.innerHTML = "

Tu carrito está vacío.

"; } else { lista.innerHTML = carrito.map((producto, indice) => `

${escaparHTML(producto.nombre)}

Subtotal: $${formatoDinero( producto.precio * producto.cantidad )}

${producto.detalles ? `

${escaparHTML(producto.detalles)}

` : ""}
${producto.cantidad}
`).join(""); } establecerTexto("total", formatoDinero(obtenerTotal())); establecerTexto("total-pago", formatoDinero(obtenerTotal())); actualizarPrecioPersonalizado(); } // ============================================ // PERSONALIZAR UN RAMO // ============================================ function calcularPrecioPersonalizado() { const tamano = obtenerElemento("tamano-ramo")?.value || ""; const envoltura = obtenerElemento("envoltura-ramo")?.value || ""; if (!PRECIOS_TAMANO[tamano] && tamano !== "Pequeño") { return 0; } if (!(envoltura in PRECIOS_ENVOLTURA)) { return 0; } return PRECIOS_TAMANO[tamano] + PRECIOS_ENVOLTURA[envoltura]; } function actualizarPrecioPersonalizado() { establecerTexto( "precio-personalizado", "$" + formatoDinero(calcularPrecioPersonalizado()) ); } function mostrarMensaje(mensaje) { const elemento = obtenerElemento("mensaje-personalizacion"); if (elemento) { elemento.textContent = mensaje; } } function inicializarPersonalizacion() { const formulario = obtenerElemento("formulario-personalizacion"); if (!formulario) return; ["tipo-flores", "color-flores", "tamano-ramo", "envoltura-ramo"] .forEach(id => { obtenerElemento(id)?.addEventListener("change", () => { actualizarPrecioPersonalizado(); mostrarMensaje(""); }); }); formulario.addEventListener("submit", evento => { evento.preventDefault(); const flores = obtenerElemento("tipo-flores").value; const color = obtenerElemento("color-flores").value; const tamano = obtenerElemento("tamano-ramo").value; const envoltura = obtenerElemento("envoltura-ramo").value; const dedicatoria = obtenerElemento("mensaje-ramo").value.trim(); if (!flores || !color || !tamano || !envoltura) { mostrarMensaje("Completa todas las opciones obligatorias."); return; } const precio = calcularPrecioPersonalizado(); const detalles = [ `Flores: ${flores}`, `Color: ${color}`, `Tamaño: ${tamano}`, `Envoltura: ${envoltura}`, dedicatoria ? `Dedicatoria: ${dedicatoria}` : "" ].filter(Boolean).join(" · "); carrito.push({ nombre: `Ramo personalizado de ${flores}`, precio, cantidad: 1, personalizado: true, detalles }); actualizarCarrito(); mostrarMensaje("¡Tu ramo personalizado se agregó al carrito!"); formulario.reset(); actualizarPrecioPersonalizado(); }); actualizarPrecioPersonalizado(); } // ============================================ // VENTANAS MODALES // ============================================ function abrirModal(id) { const modal = obtenerElemento(id); if (!modal) return; modal.classList.add("activo"); modal.setAttribute("aria-hidden", "false"); document.body.classList.add("modal-abierto"); const primerCampo = modal.querySelector( "input:not([type='hidden']), select, textarea, button" ); primerCampo?.focus(); } function cerrarModal(id) { const modal = obtenerElemento(id); if (!modal) return; modal.classList.remove("activo"); modal.setAttribute("aria-hidden", "true"); if (!document.querySelector(".modal.activo")) { document.body.classList.remove("modal-abierto"); } } function realizarPedido() { if (carrito.length === 0) { alert("Tu carrito está vacío. Agrega al menos un producto."); return; } actualizarCarrito(); mostrarDatosPago(); abrirModal("ventana-pago"); } function cerrarFormulario() { cerrarModal("ventana-pago"); } function cerrarConfirmacion() { cerrarModal("ventana-confirmacion"); } function mostrarDatosPago() { const formaPago = obtenerElemento("pago")?.value || ""; const transferencia = obtenerElemento("datos-transferencia"); const tarjeta = obtenerElemento("datos-tarjeta"); if (transferencia) { transferencia.hidden = formaPago !== "Transferencia"; } if (tarjeta) { tarjeta.hidden = formaPago !== "Tarjeta"; } } // ============================================ // CONFIRMAR PEDIDO DE DEMOSTRACIÓN // ============================================ function inicializarPedido() { const formulario = obtenerElemento("formulario-pedido"); if (!formulario) return; formulario.addEventListener("submit", evento => { evento.preventDefault(); if (carrito.length === 0) { alert("Tu carrito está vacío."); cerrarFormulario(); return; } const nombre = obtenerElemento("nombre").value.trim(); const telefono = obtenerElemento("telefono").value.trim(); const direccion = obtenerElemento("direccion").value.trim(); const pago = obtenerElemento("pago").value; if (!nombre || !direccion || !/^\d{10}$/.test(telefono) || !pago) { alert("Revisa los datos.     ` : ""}

No se verificó ninguna transferencia."; } else if (pago === "Tarjeta") { datosPago.textContent = "Pago de tarjeta simulado. No se procesó ninguna transacción."; } else { datosPago.textContent = "Método seleccionado para esta demostración."; } } cerrarFormulario(); abrirModal("ventana-confirmacion"); // No se envía información a ningún servidor. // El carrito se vacía al terminar la demostración. carrito.length = 0; actualizarCarrito(); formulario.reset(); mostrarDatosPago(); }); } // ============================================ // INICIALIZACIÓN // ============================================ document.addEventListener("DOMContentLoaded", () => { inicializarPersonalizacion(); inicializarPedido(); mostrarDatosPago(); actualizarCarrito(); obtenerElemento("cerrar-formulario")?.addEventListener( "click", cerrarFormulario ); obtenerElemento("cerrar-confirmacion")?.addEventListener( "click", cerrarConfirmacion ); obtenerElemento("boton-finalizar")?.addEventListener( "click", cerrarConfirmacion ); document.querySelectorAll(".modal").forEach(modal => { modal.addEventListener("click", evento => { if (evento.target === modal) { cerrarModal(modal.id); } }); }); document.addEventListener("keydown", evento => { if (evento.key === "Escape") { cerrarFormulario(); cerrarConfirmacion(); } }); }); 2.
